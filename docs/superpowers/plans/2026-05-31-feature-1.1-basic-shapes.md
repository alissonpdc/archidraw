# Feature 1.1: Basic Shapes — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement canvas-based shape creation for 7 shape types (rectangle, circle, diamond, triangle, line, arrow, text) with toolbar buttons, keyboard shortcuts, default styling, and JSON export/import.

**Architecture:** React + Canvas2D with Zustand state management. Shapes are pure data (immutable); a `requestAnimationFrame` render loop reads from the store and draws to canvas. Toolbar/UI in React, rendering in Canvas2D. Default styling applied by renderer, not stored per-shape.

**Tech Stack:** React 18, TypeScript, Canvas2D API, Zustand, UUID library

import { create } from 'zustand';
import { Shape, ShapeType } from '../types/shape';

interface CanvasState {
  shapes: Shape[];
  currentTool: ShapeType;
  viewport: {
    scale: number;
    offsetX: number;
    offsetY: number;
  };
  addShape: (shape: Shape) => void;
  setCurrentTool: (tool: ShapeType) => void;
  clearShapes: () => void;
  setShapes: (shapes: Shape[]) => void;
  setViewport: (scale: number, offsetX: number, offsetY: number) => void;
}

export const useCanvasStore = create<CanvasState>((set) => ({
  shapes: [],
  currentTool: 'rectangle',
  viewport: {
    scale: 1,
    offsetX: 0,
    offsetY: 0,
  },
  addShape: (shape: Shape) =>
    set((state) => ({
      shapes: [...state.shapes, shape],
    })),
  setCurrentTool: (tool: ShapeType) =>
    set(() => ({
      currentTool: tool,
    })),
  clearShapes: () =>
    set(() => ({
      shapes: [],
    })),
  setShapes: (shapes: Shape[]) =>
    set(() => ({
      shapes,
    })),
  setViewport: (scale: number, offsetX: number, offsetY: number) =>
    set(() => ({
      viewport: { scale, offsetX, offsetY },
    })),
}));
```

import { Shape } from '../types/shape';

const STROKE_COLOR = '#000000';
const STROKE_WIDTH = 2;
const FILL_COLOR = '#FFFFFF';

export class ShapeRenderer {
  private canvas: HTMLCanvasElement;
  private ctx: CanvasRenderingContext2D;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d')!;
  }

  clear(): void {
    this.ctx.fillStyle = '#FFFFFF';
    this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
  }

  drawShape(shape: Shape, scale: number, offsetX: number, offsetY: number): void {
    this.ctx.save();

    const x = shape.x * scale + offsetX;
    const y = shape.y * scale + offsetY;
    const w = shape.width * scale;
    const h = shape.height * scale;

    this.ctx.strokeStyle = shape.strokeColor;
    this.ctx.lineWidth = shape.strokeWidth;

    switch (shape.type) {
      case 'rectangle':
        this.drawRectangle(x, y, w, h, shape.fillColor);
        break;
      case 'circle':
        this.drawCircle(x + w / 2, y + h / 2, Math.min(w, h) / 2, shape.fillColor);
        break;
      case 'diamond':
        this.drawDiamond(x, y, w, h, shape.fillColor);
        break;
      case 'triangle':
        this.drawTriangle(x, y, w, h, shape.fillColor);
        break;
      case 'line':
        this.drawLine(x, y, x + w, y + h);
        break;
      case 'arrow':
        this.drawArrow(x, y, x + w, y + h);
        break;
      case 'text':
        this.drawText(x, y, shape.text);
        break;
    }

    this.ctx.restore();
  }

  private drawRectangle(x: number, y: number, w: number, h: number, fillColor: string | null): void {
    if (fillColor) {
      this.ctx.fillStyle = fillColor;
      this.ctx.fillRect(x, y, w, h);
    }
    this.ctx.strokeRect(x, y, w, h);
  }

  private drawCircle(cx: number, cy: number, r: number, fillColor: string | null): void {
    this.ctx.beginPath();
    this.ctx.arc(cx, cy, r, 0, Math.PI * 2);
    if (fillColor) {
      this.ctx.fillStyle = fillColor;
      this.ctx.fill();
    }
    this.ctx.stroke();
  }

  private drawDiamond(x: number, y: number, w: number, h: number, fillColor: string | null): void {
    const centerX = x + w / 2;
    const centerY = y + h / 2;

    this.ctx.beginPath();
    this.ctx.moveTo(centerX, y);
    this.ctx.lineTo(x + w, centerY);
    this.ctx.lineTo(centerX, y + h);
    this.ctx.lineTo(x, centerY);
    this.ctx.closePath();

    if (fillColor) {
      this.ctx.fillStyle = fillColor;
      this.ctx.fill();
    }
    this.ctx.stroke();
  }

  private drawTriangle(x: number, y: number, w: number, h: number, fillColor: string | null): void {
    this.ctx.beginPath();
    this.ctx.moveTo(x + w / 2, y);
    this.ctx.lineTo(x + w, y + h);
    this.ctx.lineTo(x, y + h);
    this.ctx.closePath();

    if (fillColor) {
      this.ctx.fillStyle = fillColor;
      this.ctx.fill();
    }
    this.ctx.stroke();
  }

  private drawLine(x1: number, y1: number, x2: number, y2: number): void {
    this.ctx.beginPath();
    this.ctx.moveTo(x1, y1);
    this.ctx.lineTo(x2, y2);
    this.ctx.stroke();
  }

  private drawArrow(x1: number, y1: number, x2: number, y2: number): void {
    const headlen = 15;
    const angle = Math.atan2(y2 - y1, x2 - x1);

    this.ctx.beginPath();
    this.ctx.moveTo(x1, y1);
    this.ctx.lineTo(x2, y2);
    this.ctx.stroke();

    this.ctx.beginPath();
    this.ctx.moveTo(x2, y2);
    this.ctx.lineTo(x2 - headlen * Math.cos(angle - Math.PI / 6), y2 - headlen * Math.sin(angle - Math.PI / 6));
    this.ctx.lineTo(x2 - headlen * Math.cos(angle + Math.PI / 6), y2 - headlen * Math.sin(angle + Math.PI / 6));
    this.ctx.closePath();
    this.ctx.fill();
  }

  private drawText(x: number, y: number, text: string): void {
    this.ctx.fillStyle = '#000000';
    this.ctx.font = '14px system-ui, sans-serif';
    this.ctx.fillText(text || 'Text', x, y + 14);
  }
}

import { v4 as uuidv4 } from 'uuid';
import { Shape, ShapeType } from '../types/shape';

export function createShape(
  type: ShapeType,
  x: number,
  y: number,
  width: number,
  height: number,
  text: string = ''
): Shape {
  const isFillable = type !== 'line' && type !== 'arrow';

  return {
    id: uuidv4(),
    type,
    x,
    y,
    width: Math.max(0, width),
    height: Math.max(0, height),
    rotation: 0,
    text,
    strokeColor: '#000000',
    strokeWidth: 2,
    fillColor: isFillable ? '#FFFFFF' : null,
    createdAt: new Date().toISOString(),
  };
}

export function calculateAspectRatio(width: number, height: number): number {
  return width / (height || 1);
}

export function maintainAspectRatio(
  width: number,
  height: number,
  type: ShapeType
): { width: number; height: number } {
  // Circle: always square
  if (type === 'circle') {
    const size = Math.min(Math.abs(width), Math.abs(height));
    return { width: size, height: size };
  }

  // Rectangle: square
  if (type === 'rectangle') {
    const size = Math.min(Math.abs(width), Math.abs(height));
    return { width: size, height: size };
  }

  // Triangle, Diamond: square aspect
  if (type === 'triangle' || type === 'diamond') {
    const size = Math.min(Math.abs(width), Math.abs(height));
    return { width: size, height: size };
  }

  // Line, Arrow, Text: no aspect ratio constraint
  return { width, height };
}
```

import React, { useEffect, useRef, useState } from 'react';
import { useCanvasStore } from '../store/canvasStore';
import { ShapeRenderer } from '../utils/shapeRenderer';
import { createShape, maintainAspectRatio } from '../utils/shapeFactory';

export const Canvas: React.FC = () => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const { shapes, currentTool, viewport, addShape } = useCanvasStore();
  const [isDrawing, setIsDrawing] = useState(false);
  const [startX, setStartX] = useState(0);
  const [startY, setStartY] = useState(0);
  const rendererRef = useRef<ShapeRenderer | null>(null);

  // Initialize renderer
  useEffect(() => {
    if (!canvasRef.current) return;
    const canvas = canvasRef.current;
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight - 50; // Leave space for toolbar
    rendererRef.current = new ShapeRenderer(canvas);
  }, []);

  // Render loop
  useEffect(() => {
    if (!rendererRef.current || !canvasRef.current) return;

    let animationId: number;

    const render = () => {
      rendererRef.current!.clear();
      shapes.forEach((shape) => {
        rendererRef.current!.drawShape(shape, viewport.scale, viewport.offsetX, viewport.offsetY);
      });
      animationId = requestAnimationFrame(render);
    };

    animationId = requestAnimationFrame(render);
    return () => cancelAnimationFrame(animationId);
  }, [shapes, viewport]);

  const handleMouseDown = (e: React.MouseEvent<HTMLCanvasElement>) => {
    const rect = canvasRef.current!.getBoundingClientRect();
    const x = (e.clientX - rect.left - viewport.offsetX) / viewport.scale;
    const y = (e.clientY - rect.top - viewport.offsetY) / viewport.scale;
    setStartX(x);
    setStartY(y);
    setIsDrawing(true);
  };

  const handleMouseMove = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (!isDrawing) return;

    const rect = canvasRef.current!.getBoundingClientRect();
    const x = (e.clientX - rect.left - viewport.offsetX) / viewport.scale;
    const y = (e.clientY - rect.top - viewport.offsetY) / viewport.scale;

    // Preview: could be enhanced with ghost rendering
  };

  const handleMouseUp = (e: React.MouseEvent<HTMLCanvasElement>) => {
    if (!isDrawing) return;
    setIsDrawing(false);

    const rect = canvasRef.current!.getBoundingClientRect();
    const endX = (e.clientX - rect.left - viewport.offsetX) / viewport.scale;
    const endY = (e.clientY - rect.top - viewport.offsetY) / viewport.scale;

    const width = endX - startX;
    const height = endY - startY;

    // Ignore tiny clicks
    if (Math.abs(width) < 2 || Math.abs(height) < 2) return;

    const shapeX = width < 0 ? endX : startX;
    const shapeY = height < 0 ? endY : startY;
    let shapeWidth = Math.abs(width);
    let shapeHeight = Math.abs(height);

    // Apply Shift constraint if pressed
    if (e.shiftKey) {
      const { width: w, height: h } = maintainAspectRatio(shapeWidth, shapeHeight, currentTool);
      shapeWidth = w;
      shapeHeight = h;
    }

    const shape = createShape(currentTool, shapeX, shapeY, shapeWidth, shapeHeight);
    addShape(shape);
  };

  const handleKeyDown = (e: KeyboardEvent) => {
    const toolMap: Record<string, any> = {
      r: 'rectangle',
      c: 'circle',
      d: 'diamond',
      t: 'triangle',
      l: 'line',
      a: 'arrow',
      x: 'text',
    };
    const tool = toolMap[e.key.toLowerCase()];
    if (tool) {
      useCanvasStore.setState({ currentTool: tool });
    }
  };

  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  return (
    <canvas
      ref={canvasRef}
      onMouseDown={handleMouseDown}
      onMouseMove={handleMouseMove}
      onMouseUp={handleMouseUp}
      onMouseLeave={() => setIsDrawing(false)}

        cursor: isDrawing ? 'crosshair' : 'default',
      }}
    />
  );
};

import { Shape, CanvasDocument } from '../types/shape';

export function exportToJSON(shapes: Shape[]): string {
  const doc: CanvasDocument = {
    version: '1.0',
    date: new Date().toISOString(),
    shapes,
  };
  return JSON.stringify(doc, null, 2);
}

export function downloadFile(content: string, filename: string): void {
  const blob = new Blob([content], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}
```

import { Shape, CanvasDocument } from '../types/shape';

export function importFromJSON(jsonString: string): Shape[] {
  try {
    const doc: CanvasDocument = JSON.parse(jsonString);
    if (!doc.shapes || !Array.isArray(doc.shapes)) {
      throw new Error('Invalid .archidraw format: missing shapes array');
    }
    return doc.shapes;
  } catch (error) {
    throw new Error(`Failed to parse .archidraw file: ${error}`);
  }
}

export function readFileAsText(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = (e) => resolve(e.target?.result as string);
    reader.onerror = () => reject(reader.error);
    reader.readAsText(file);
  });
}
```
